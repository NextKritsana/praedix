```jsx
// src/App.jsx
import { BrowserRouter, Routes, Route, Navigate } from "react-router-dom";
import NavBar from "./components/NavBar.jsx";
import Landing from "./components/Landing.jsx";
import LoginTailwind from "./components/LoginTailwind.jsx";
import SignUp from "./components/SignUp.jsx";
import Dashboard from "./components/Dashboard.jsx";
import UploadEvidence from "./components/UploadEvidence.jsx";
import ViewCases from "./components/ViewCases.jsx";
import CaseDetail from "./components/CaseDetail.jsx";

// Simple auth check
const isAuthenticated = () => {
  return localStorage.getItem("auth_token") !== null;
};

// Protected Route component
const ProtectedRoute = ({ children }) => {
  return isAuthenticated() ? children : <Navigate to="/login" />;
};

function App() {
  return (
    <BrowserRouter>
      <Routes>
        {/* Public routes with navbar */}
        <Route path="/" element={
          <>
            <NavBar />
            <Landing />
          </>
        } />
        
        {/* Auth routes without navbar */}
        <Route path="/login" element={<LoginTailwind />} />
        <Route path="/signup" element={<SignUp />} />
        
        {/* Protected routes WITH navbar - components don't have their own headers anymore */}
        <Route path="/dashboard" element={
          <ProtectedRoute>
            <>
              <NavBar />
              <Dashboard />
            </>
          </ProtectedRoute>
        } />
        
        <Route path="/upload" element={
          <ProtectedRoute>
            <>
              <NavBar />
              <UploadEvidence />
            </>
          </ProtectedRoute>
        } />

        <Route path="/cases" element={
          <ProtectedRoute>
            <>
              <NavBar />
              <ViewCases />
            </>
          </ProtectedRoute>
        } />

        {/* Case Detail route */}
        <Route path="/cases/:caseId" element={
          <ProtectedRoute>
            <>
              <NavBar />
              <CaseDetail />
            </>
          </ProtectedRoute>
        } />

        {/* Reports route (if you create this later) */}
        <Route path="/reports" element={
          <ProtectedRoute>
            <>
              <NavBar />
              <div className="min-h-screen bg-slate-50 py-8">
                <div className="max-w-4xl mx-auto px-4">
                  <div className="bg-white rounded-2xl shadow-lg p-6 text-center">
                    <h1 className="text-2xl font-bold text-slate-900 mb-4">Reports</h1>
                    <p className="text-slate-600">Reports feature coming soon...</p>
                  </div>
                </div>
              </div>
            </>
          </ProtectedRoute>
        } />
        
        {/* Redirect unknown routes to home */}
        <Route path="*" element={<Navigate to="/" />} />
      </Routes>
    </BrowserRouter>
  );
}

export default App;
```
